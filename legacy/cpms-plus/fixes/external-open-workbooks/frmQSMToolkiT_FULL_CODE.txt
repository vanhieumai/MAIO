Private Sub UpdateSourceLabel()

Private Sub btnExternalSource_Click()
    On Error GoTo EH
    Dim fd As FileDialog, wbSrc As Workbook, wbTarget As Workbook, wb As Workbook
    Dim selectedPath As String, candidates As Collection
    Dim picker As frmQSMSourcePicker, answer As VbMsgBoxResult
    Dim oldScreen As Boolean, oldLinks As Boolean, settingsChanged As Boolean
    Set wbTarget = defaultWorkbook
    If wbTarget Is Nothing Then
        ShowMsgCenter "No target workbook loaded.", vbExclamation
        Exit Sub
    End If
    Set candidates = New Collection
    For Each wb In Application.Workbooks
        If Not (wb Is wbTarget) Then
            If Not wb.IsAddin And UCase$(wb.Name) <> "PERSONAL.XLSB" Then
                If wb.Windows.count > 0 Then
                    If wb.Windows(1).Visible Then candidates.Add wb
                End If
            End If
        End If
    Next wb
    If candidates.count > 0 Then
        answer = MsgBox("Load Source data from an Excel workbook already open?" & vbCrLf & _
                        "Yes: use an open workbook. No: browse for a file.", _
                        vbYesNoCancel + vbQuestion, "Load External")
        If answer = vbCancel Then Exit Sub
        If answer = vbYes Then
            If candidates.count = 1 Then
                Set wbSrc = candidates(1)
            Else
                Set picker = New frmQSMSourcePicker
                picker.Configure candidates
                picker.Show vbModal
                Set wbSrc = picker.SelectedWorkbook
                Unload picker
                Set picker = Nothing
                If wbSrc Is Nothing Then Exit Sub
            End If
        End If
    End If
    If wbSrc Is Nothing Then
        Set fd = Application.FileDialog(msoFileDialogFilePicker)
        With fd
            .title = "Select source Excel file"
            .Filters.Clear
            .Filters.Add "Excel Files", "*.xlsx; *.xlsm"
            .AllowMultiSelect = False
            If .Show <> -1 Then Exit Sub
        End With
        selectedPath = fd.SelectedItems(1)
        For Each wb In Application.Workbooks
            If StrComp(wb.fullName, selectedPath, vbTextCompare) = 0 Then
                Set wbSrc = wb
                Exit For
            End If
        Next wb
        If wbSrc Is Nothing Then
            oldScreen = Application.ScreenUpdating
            oldLinks = Application.AskToUpdateLinks
            settingsChanged = True
            Application.ScreenUpdating = False
            Application.AskToUpdateLinks = False
            Set wbSrc = Workbooks.Open(fileName:=selectedPath, UpdateLinks:=0, _
                         ReadOnly:=True, IgnoreReadOnlyRecommended:=True, AddToMru:=False)
            Application.AskToUpdateLinks = oldLinks
            Application.ScreenUpdating = oldScreen
            settingsChanged = False
        End If
    End If
    If wbSrc Is wbTarget Then
        ShowMsgCenter "Please choose a Source workbook different from the Target.", vbExclamation
        Exit Sub
    End If
    Set srcWorkbook = wbSrc
    srcWbName = wbSrc.Name
    srcWbPath = wbSrc.fullName
    isUsingExternalSource = True
    cbCotTenTH.Clear
    lbCotKL_TH.Clear
    LoadSourceSheets
    UpdateSourceLabel
    wbTarget.Activate
    ShowMsgCenter "External file loaded as Source: " & wbSrc.Name
    Exit Sub
EH:
    Dim message As String
    message = Err.Description
    On Error Resume Next
    If settingsChanged Then
        Application.AskToUpdateLinks = oldLinks
        Application.ScreenUpdating = oldScreen
    End If
    If Not picker Is Nothing Then Unload picker
    ShowMsgCenter "Cannot load source file: " & message, vbExclamation
End Sub

Private Function FindMatchingRow(wsTH As Worksheet, wsGT As Worksheet, _
                                 rGT As Long, colTenTH As Long, colTenGT As Long) As Long
    Dim i As Long, rTH As Long
    Dim hm As String
    Dim countGT As Long, countTH As Long
    Dim firstRow As Long: firstRow = 10

    hm = Trim(wsGT.Cells(rGT, colTenGT).value)

    For i = firstRow To rGT
        If Trim(wsGT.Cells(i, colTenGT).value) = hm Then countGT = countGT + 1
    Next i

    For rTH = firstRow To wsTH.Cells(wsTH.rows.count, colTenTH).End(xlUp).Row
        If Trim(wsTH.Cells(rTH, colTenTH).value) = hm Then
            countTH = countTH + 1
            If countTH = countGT Then
                FindMatchingRow = rTH
                Exit Function
            End If
        End If
    Next rTH
End Function

Private Function FormatPreviewValue(v As Variant) As String

    ' 1. Error value (#N/A, #VALUE!, ...)
    If IsError(v) Then
        FormatPreviewValue = ""
        Exit Function
    End If

    ' 2. Empty / Null
    If IsEmpty(v) Or IsNull(v) Then
        FormatPreviewValue = ""
        Exit Function
    End If

    ' 3. Empty string
    If Trim(CStr(v)) = "" Then
        FormatPreviewValue = ""
        Exit Function
    End If

    ' 4. Numeric
    If IsNumeric(v) Then
        FormatPreviewValue = Format(CDbl(v), "#,##0.00")
        Exit Function
    End If

    ' 5. Default
    FormatPreviewValue = CStr(v)

End Function



Private Function ValidateInputForLink() As Boolean

    Dim i As Long
    Dim cntTH As Long, cntGT As Long

    '--- CHECK SHEET
    If cbSheetTH.value = "" Or cbSheetGT.value = "" Then
        ShowMsgCenter "Please select source sheet and target sheet", vbExclamation
        Exit Function
    End If

    '--- CHECK C?T TÊN
    If cbCotTenTH.value = "" Or cbCotTenGT.value = "" Then
        ShowMsgCenter "Please select matching column", vbCritical
        Exit Function
    End If

    '--- COUNT C?T KL
    For i = 0 To lbCotKL_TH.ListCount - 1
        If lbCotKL_TH.Selected(i) Then cntTH = cntTH + 1
    Next i

    For i = 0 To lbCotKL_GT.ListCount - 1
        If lbCotKL_GT.Selected(i) Then cntGT = cntGT + 1
    Next i

    If cntTH = 0 Or cntGT = 0 Then
        ShowMsgCenter "Please select the data column", vbCritical
        Exit Function
    End If

    If cntTH <> cntGT Then
        ShowMsgCenter "The number of data columns in the source and target must be the same", vbExclamation
        Exit Function
    End If

    ValidateInputForLink = True

End Function

'==============================
' LOAD COLUMNS (SAFE – NO TYPE ERROR)
'==============================
Private Sub LoadColumns(ws As Worksheet, cbo As Object, lb As Object, _
                        Optional extraCols As Long = 0)

    Dim i As Long
    Dim lastCol As Long

    cbo.Clear
    lb.Clear
    lb.MultiSelect = fmMultiSelectMulti

    '=== TÌM C?T CU?I CÓ DATA (B?T K? ROW NÀO) ===
    ' Dùng Find thay vì UsedRange (tránh th?a do ô có format không có value)
    ' và thay vì End(xlToLeft) trên row 1 (b? sót khi row 1 tr?ng)
    Dim lastCell As Range
    On Error Resume Next
    Set lastCell = ws.Cells.Find(What:="*", _
                                 After:=ws.Cells(1, 1), _
                                 LookIn:=xlFormulas, _
                                 SearchOrder:=xlByColumns, _
                                 SearchDirection:=xlPrevious)
    On Error GoTo 0

    If lastCell Is Nothing Then
        If extraCols = 0 Then Exit Sub
        lastCol = 0
    Else
        lastCol = lastCell.Column
    End If

    For i = 1 To lastCol + extraCols
        cbo.AddItem ws.Cells(1, i).Address(False, False)
        lb.AddItem ws.Cells(1, i).Address(False, False)
    Next i

End Sub

Private Function HasAnyContract(ws As Worksheet) As Boolean
    Dim r As Long
    For r = 2 To ws.Cells(ws.rows.count, 1).End(xlUp).Row
        If Trim(ws.Cells(r, 1).value) <> "" _
           And Left(ws.Cells(r, 1).value, 5) <> "GROUP" Then
            HasAnyContract = True
            Exit Function
        End If
    Next r
    HasAnyContract = False
End Function




Public Sub SyncLedgerFromContract(contractID As String)

    Dim wsLedger As Worksheet
    Dim wsDB As Worksheet
    Dim r As Long, lastRow As Long
    Dim dbRow As Long

    If Trim(contractID) = "" Then Exit Sub

    Set wsLedger = EnsurePaymentLedger
    Set wsDB = EnsureContractDB

    ' === tìm contract trong DB ===
    dbRow = FindContractRowByID(wsDB, contractID)
    If dbRow = 0 Then Exit Sub

    lastRow = wsLedger.Cells(wsLedger.rows.count, "C").End(xlUp).Row

    ' === quét toàn b? ledger ===
    For r = 2 To lastRow

        ' ===== dúng contract family =====
        If (Trim(wsLedger.Cells(r, "C").value) = contractID) And (Trim(wsLedger.Cells(r, "B").value) = "") Then

            With wsLedger

                ' ===== CORE INFO =====
                Call WriteTextCell(.Cells(r, "D"), wsDB.Cells(dbRow, "B").value)
                .Cells(r, "F").value = wsDB.Cells(dbRow, "K").value   ' Scope / ItemName

                ' ===== FINANCIAL MASTER =====
                .Cells(r, "G").value = wsDB.Cells(dbRow, "M").value   ' ExchangeRate
                .Cells(r, "H").value = wsDB.Cells(dbRow, "N").value   ' EstimatedValue
                .Cells(r, "I").value = wsDB.Cells(dbRow, "O").value   ' ContractValue

            End With

        End If

    Next r

End Sub





Private Function LinkInputsReady() As Boolean
    Dim i As Long, a As Long, b As Long
    If cbSheetTH.ListIndex < 0 Or cbSheetGT.ListIndex < 0 Or cbCotTenTH.ListIndex < 0 Or cbCotTenGT.ListIndex < 0 Then Exit Function
    For i = 0 To lbCotKL_TH.ListCount - 1
        If lbCotKL_TH.Selected(i) Then a = a + 1
    Next i
    For i = 0 To lbCotKL_GT.ListCount - 1
        If lbCotKL_GT.Selected(i) Then b = b + 1
    Next i
    LinkInputsReady = (a > 0 And a = b)
End Function

Private Function SelectedLinkColumns(ByVal lb As Object, ByVal ws As Worksheet) As String
    Dim i As Long, result As String
    For i = 0 To lb.ListCount - 1
        If lb.Selected(i) Then
            If result <> "" Then result = result & ","
            result = result & ws.Range(CStr(lb.List(i))).Column
        End If
    Next i
    SelectedLinkColumns = result
End Function

Private Function CaptureLinkPlan() As clsQSMLinkPlan
    If Not LinkInputsReady() Then Err.Raise 5, , "Select both sheets, matching columns and the same number of link columns."
    If Not IsExternalSourceAlive() Then Err.Raise 5, , "Source workbook has been closed."
    Dim source As Worksheet, target As Worksheet, wb As Workbook, plan As clsQSMLinkPlan
    If srcWorkbook Is Nothing Then Set wb = defaultWorkbook Else Set wb = srcWorkbook
    Set source = wb.Worksheets(CStr(cbSheetTH.value))
    Set target = defaultWorkbook.Worksheets(CStr(cbSheetGT.value))
    Set plan = New clsQSMLinkPlan
    plan.Configure source, target, source.Range(CStr(cbCotTenTH.value)).Column, target.Range(CStr(cbCotTenGT.value)).Column, _
                   SelectedLinkColumns(lbCotKL_TH, source), SelectedLinkColumns(lbCotKL_GT, target)
    Set CaptureLinkPlan = plan
End Function

Private Sub RememberLinkSetup()
    ' Only UI configuration is stored; no worksheet contents or credentials.
    Dim source As Workbook, record As String, token As String, i As Long, old As String
    On Error GoTo Failed
    If srcWorkbook Is Nothing Then Set source = defaultWorkbook Else Set source = srcWorkbook
    If source.path = "" Or defaultWorkbook.path = "" Then token = mHistorySession
    record = source.fullName & vbTab & defaultWorkbook.fullName & vbTab & CStr(cbSheetTH.value) & vbTab & _
        CStr(cbSheetGT.value) & vbTab & CStr(cbCotTenTH.value) & vbTab & CStr(cbCotTenGT.value) & vbTab & _
        SelectedLinkColumns(lbCotKL_TH, source.Worksheets(CStr(cbSheetTH.value))) & vbTab & _
        SelectedLinkColumns(lbCotKL_GT, defaultWorkbook.Worksheets(CStr(cbSheetGT.value))) & vbTab & token
    If token <> "" Then
        If mUnsavedHistory Is Nothing Then Set mUnsavedHistory = CreateObject("Scripting.Dictionary")
        Dim pair As Collection
        Set pair = New Collection
        pair.Add source
        pair.Add defaultWorkbook
        If mUnsavedHistory.exists(record) Then mUnsavedHistory.Remove record
        mUnsavedHistory.Add record, pair
    End If
    Dim records As Collection
    Set records = New Collection
    records.Add record
    For i = 1 To 5
        old = GetSetting("CPMS_QSMToolkit", "LinkHistoryV1", CStr(i), "")
        If old <> "" And old <> record And records.count < 5 Then records.Add old
    Next i
    For i = 1 To records.count
        SaveSetting "CPMS_QSMToolkit", "LinkHistoryV1", CStr(i), CStr(records(i))
    Next i
    Exit Sub
Failed:
    MsgBox "Preview/link can continue, but history could not be saved: " & Err.Description, vbExclamation, "QSM history"
End Sub

Private Sub RestoreHistoryColumns(ByVal ws As Worksheet, ByVal cbo As Object, ByVal lb As Object, ByVal matchAddress As String, ByVal columns As String)
    Dim parts As Variant, item As Variant, maxCol As Long, n As Long, i As Long
    parts = Split(columns, ",")
    maxCol = ws.Range(matchAddress).Column
    For Each item In parts
        n = CLng(item)
        If n < 1 Or n > ws.columns.count Then Err.Raise 5, , "History contains an invalid column."
        If n > maxCol Then maxCol = n
    Next item
    LoadColumns ws, cbo, lb, 1
    For i = lb.ListCount + 1 To maxCol
        cbo.AddItem ws.Cells(1, i).Address(False, False)
        lb.AddItem ws.Cells(1, i).Address(False, False)
    Next i
    cbo.value = matchAddress
    For Each item In parts
        lb.Selected(CLng(item) - 1) = True
    Next item
End Sub

Private Function RestoreLinkHistory() As Boolean
    Dim records As Collection, i As Long, record As String, fields As Variant
    Dim menu As String, answer As Variant, idx As Long
    Dim source As Workbook, target As Workbook, sourceSheet As Worksheet, targetSheet As Worksheet, ws As Worksheet
    On Error GoTo Failed
    Set records = New Collection
    For i = 1 To 5
        record = GetSetting("CPMS_QSMToolkit", "LinkHistoryV1", CStr(i), "")
        If record <> "" Then
            fields = Split(record, vbTab)
            If UBound(fields) = 8 Then
                records.Add record
                menu = menu & records.count & ". " & Left$(fields(2), 25) & " -> " & Left$(fields(3), 25) & _
                    " [" & Left$(mID$(fields(1), InStrRev(fields(1), "\") + 1), 30) & "]" & vbCrLf
            End If
        End If
    Next i
    If records.count = 0 Then
        MsgBox "No history yet. Select the sheets/columns manually once and open Preview or Link Data to save the setup.", vbInformation, "QSM history"
        Exit Function
    End If
    answer = InputBox("Input is incomplete. Choose a saved setup by number:" & vbCrLf & menu, "QSM history")
    If Len(Trim$(CStr(answer))) = 0 Then Exit Function
    If Not IsNumeric(answer) Then Err.Raise 5, , "Enter a history number."
    answer = CDbl(answer)
    If answer < 1 Or answer > records.count Or answer <> Fix(answer) Then Err.Raise 5, , "Invalid history number."
    idx = CLng(answer)
    fields = Split(CStr(records(idx)), vbTab)
    If fields(8) <> "" And fields(8) <> mHistorySession Then Err.Raise 5, , "This setup used an unsaved workbook in an earlier session. Select the workbook manually."
    Set source = FindHistoryWorkbook(CStr(fields(0)))
    Set target = FindHistoryWorkbook(CStr(fields(1)))
    If source Is Nothing Or target Is Nothing Then Err.Raise 5, , "Open the original source and target workbooks first:" & vbCrLf & fields(0) & vbCrLf & fields(1)
    If fields(8) <> "" Then
        If mUnsavedHistory Is Nothing Then Err.Raise 5, , "Unsaved workbook history is no longer available."
        If Not mUnsavedHistory.exists(CStr(records(idx))) Then Err.Raise 5, , "Unsaved workbook history is no longer available."
        Dim originals As Collection, originalSource As Workbook, originalTarget As Workbook
        Set originals = mUnsavedHistory(CStr(records(idx)))
        Set originalSource = originals(1): Set originalTarget = originals(2)
        If Not (source Is originalSource) Or Not (target Is originalTarget) Then Err.Raise 5, , "The original unsaved workbook was closed. Select the new workbook manually."
    End If
    Set sourceSheet = source.Worksheets(CStr(fields(2)))
    Set targetSheet = target.Worksheets(CStr(fields(3)))
    If sourceSheet.Visible <> xlSheetVisible Or targetSheet.Visible <> xlSheetVisible Then Err.Raise 5, , "A history worksheet is hidden. Show it before restoring."
    isBusy = True
    Set defaultWorkbook = target
    Set srcWorkbook = Nothing
    srcWbName = "": srcWbPath = "": isUsingExternalSource = False
    If Not (source Is target) Then
        Set srcWorkbook = source
        srcWbName = source.Name: srcWbPath = source.fullName: isUsingExternalSource = True
    End If
    cbSheetTH.Clear: cbSheetGT.Clear
    For Each ws In source.Worksheets
        If ws.Visible = xlSheetVisible Then cbSheetTH.AddItem ws.Name
    Next ws
    For Each ws In target.Worksheets
        If ws.Visible = xlSheetVisible Then cbSheetGT.AddItem ws.Name
    Next ws
    cbSheetTH.value = sourceSheet.Name
    cbSheetGT.value = targetSheet.Name
    RestoreHistoryColumns sourceSheet, cbCotTenTH, lbCotKL_TH, CStr(fields(4)), CStr(fields(6))
    RestoreHistoryColumns targetSheet, cbCotTenGT, lbCotKL_GT, CStr(fields(5)), CStr(fields(7))
    UpdateSourceLabel
    isBusy = False
    RestoreLinkHistory = LinkInputsReady()
    Exit Function
Failed:
    isBusy = False
    MsgBox "Cannot restore history: " & Err.Description, vbExclamation, "QSM history"
End Function
