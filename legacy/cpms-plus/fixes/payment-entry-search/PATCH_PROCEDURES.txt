Public Sub LoadPaymentHistoryFast( _
    contractID As String)

    Dim rows As Collection
    Dim idx As Variant

    Dim arrList() As Variant
    Dim finalArr() As Variant

    Dim total As Long
    Dim r As Long
    Dim c As Long

    On Error GoTo SearchFailed
    If mSearchContractID <> contractID Then
        mSearchContractID = contractID
        mPaymentSearchKeyword = ""
        txtSearch.Value = ""
    End If
    Dim previousSuppress As Boolean
    previousSuppress = suppressPaymentSelect
    ResetPaymentSelection
    selectedLedgerRow = 0
    lblLedgerID.Caption = ""
    btnDeletePayment.Enabled = False
    suppressPaymentSelect = True
    lstPayment.Clear
    If gDictLedgerByContract Is Nothing Then GoTo SearchDone

    If Not gDictLedgerByContract.exists(contractID) Then GoTo SearchDone

    Set rows = gDictLedgerByContract(contractID)

    ReDim arrList(0 To rows.count + 5, 0 To 8)

    total = 0

    For Each idx In rows

        ' PAYMENT ROW ONLY
        If Trim(CStr(gArrLedger(idx, 2))) <> "" Then

            arrList(total, 0) = idx

            arrList(total, 1) = gArrLedger(idx, 16)

            arrList(total, 2) = gArrLedger(idx, 15)

            arrList(total, 3) = _
                FastVN(gArrLedger(idx, 10))

            arrList(total, 4) = _
                FastVN(gArrLedger(idx, 11))

            arrList(total, 5) = _
                FastVN(gArrLedger(idx, 12))

            arrList(total, 6) = _
                FastVN(gArrLedger(idx, 13))

            arrList(total, 7) = _
                FastVN(gArrLedger(idx, 14))

            If IsDate(gArrLedger(idx, 17)) Then

                arrList(total, 8) = _
                    Format(gArrLedger(idx, 17), _
                    "dd/mm/yyyy")

            End If

            total = total + 1

        End If

    Next idx

    If total > 0 Then
    
        ReDim finalArr(0 To total - 1, 0 To 8)
    
        For r = 0 To total - 1
    
            finalArr(r, 0) = arrList(r, 0)
            finalArr(r, 1) = arrList(r, 1)
            finalArr(r, 2) = arrList(r, 2)
            finalArr(r, 3) = arrList(r, 3)
            finalArr(r, 4) = arrList(r, 4)
            finalArr(r, 5) = arrList(r, 5)
            finalArr(r, 6) = arrList(r, 6)
            finalArr(r, 7) = arrList(r, 7)
            finalArr(r, 8) = arrList(r, 8)
    
        Next r
    
        Dim filtered() As Variant, matchCount As Long, outputRow As Long
        Dim allRows As Variant
        allRows = finalArr
        For r = 0 To total - 1
            If PaymentMatchesSearch(allRows, r) Then matchCount = matchCount + 1
        Next r
        If matchCount > 0 Then
            ReDim filtered(0 To matchCount - 1, 0 To 8)
            For r = 0 To total - 1
                If PaymentMatchesSearch(allRows, r) Then
                    For c = 0 To 8
                        filtered(outputRow, c) = allRows(r, c)
                    Next c
                    outputRow = outputRow + 1
                End If
            Next r
            lstPayment.List = filtered
        End If
    
    End If

SearchDone:
    suppressPaymentSelect = previousSuppress
    txtSearch.ControlTipText = "Enter: search; Esc or double right-click: clear. Results: " & lstPayment.ListCount
    Exit Sub
SearchFailed:
    suppressPaymentSelect = previousSuppress
    MsgBox "Cannot load payment search results: " & Err.Description, vbExclamation, "Payment Entry"
End Sub

Private Function PaymentSearchText(ByVal value As Variant) As String
    If IsError(value) Or IsNull(value) Or IsEmpty(value) Then Exit Function
    PaymentSearchText = LCase$(VN_NoSign(Trim$(CStr(value))))
End Function

Private Function PaymentMatchesSearch(ByRef data As Variant, ByVal rowIndex As Long) As Boolean
    Dim col As Long, ledgerIndex As Long
    If mPaymentSearchKeyword = "" Then PaymentMatchesSearch = True: Exit Function
    For col = 1 To 8
        If InStr(1, PaymentSearchText(data(rowIndex, col)), mPaymentSearchKeyword, vbBinaryCompare) > 0 Then
            PaymentMatchesSearch = True
            Exit Function
        End If
    Next col
    ledgerIndex = CLng(data(rowIndex, 0))
    PaymentMatchesSearch = (InStr(1, PaymentSearchText(gArrLedger(ledgerIndex, 2)), mPaymentSearchKeyword, vbBinaryCompare) > 0)
End Function

Private Sub ApplyPaymentSearch()
    mPaymentSearchKeyword = PaymentSearchText(txtSearch.Value)
    LoadPaymentHistoryFast CStr(lblContractID.Caption)
End Sub

Private Sub txtSearch_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    If KeyCode = vbKeyReturn Then
        KeyCode = 0
        ApplyPaymentSearch
    ElseIf KeyCode = vbKeyEscape Then
        KeyCode = 0
        txtSearch.Value = ""
        ApplyPaymentSearch
    End If
End Sub

Private Sub txtSearch_MouseDown(ByVal Button As Integer, ByVal Shift As Integer, ByVal X As Single, ByVal Y As Single)
    If Button <> 2 Then Exit Sub
    Dim nowTime As Double, elapsed As Double
    nowTime = Timer
    elapsed = nowTime - mSearchRightClickTime
    If elapsed < 0 Then elapsed = elapsed + 86400
    If mSearchRightClickCount > 0 And elapsed < 0.5 Then
        mSearchRightClickCount = 0
        txtSearch.Value = ""
        ApplyPaymentSearch
    Else
        mSearchRightClickCount = 1
        mSearchRightClickTime = nowTime
    End If
End Sub
