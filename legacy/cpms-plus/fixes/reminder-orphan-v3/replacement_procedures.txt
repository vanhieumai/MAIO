Private Sub CreateOrphanDeleteButton()
    Set mDeleteOrphan = Me.Controls.Add("Forms.CommandButton.1", "btnDeleteOrphan", True)
    Set mAssignContract = Me.Controls.Add("Forms.CommandButton.1", "btnAssignContract", True)
    mDeleteOrphan.Caption = "Delete orphan"
    mAssignContract.Caption = "Assign contract"
    mDeleteOrphan.Visible = False
    mAssignContract.Visible = False
    StyleOrphanButtons
End Sub

Private Sub StyleOrphanButtons()
    If mDeleteOrphan Is Nothing Then Exit Sub
    MatchReminderButton mDeleteOrphan
    MatchReminderButton mAssignContract
    mDeleteOrphan.Top = btnRefreshRiminder.Top
    mAssignContract.Top = btnGoTo.Top
    ' Keep Refresh and Go to at their designer positions.
    Dim gap As Single
    gap = btnGoTo.Left - (btnRefreshRiminder.Left + btnRefreshRiminder.Width)
    If gap < 8 Then gap = 8
    mDeleteOrphan.Left = btnRefreshRiminder.Left - gap - mDeleteOrphan.Width
    mAssignContract.Left = btnGoTo.Left + btnGoTo.Width + gap
End Sub

Private Sub UpdateReminderActions()
    Dim wb As Workbook, ws As Worksheet, rowNo As Long
    Dim docID As String, contractID As String, exists As Boolean
    CloseContractPicker
    btnGoTo.Enabled = False
    If Not mDeleteOrphan Is Nothing Then mDeleteOrphan.Visible = False
    If Not mAssignContract Is Nothing Then mAssignContract.Visible = False
    On Error GoTo Done
    If Not ResolveReminderSelection(wb, ws, rowNo, docID, contractID) Then Exit Sub
    exists = ReminderContractExists(wb, contractID)
    btnGoTo.Enabled = exists
    If Not exists And Not mDeleteOrphan Is Nothing Then
        StyleOrphanButtons
        mDeleteOrphan.Visible = True
        mAssignContract.Visible = True
        mDeleteOrphan.Enabled = Not wb.ReadOnly And Not ws.ProtectContents
        mAssignContract.Enabled = mDeleteOrphan.Enabled
    End If
Done:
End Sub
