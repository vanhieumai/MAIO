'================================================================
' NAP UNIQUE DOC TYPES TU COT C SHEET DOCUMENT_TRACKER
' (bo qua "Interim Payment Certificate" vi da co opt rieng)
'================================================================
Private Sub LoadDocTypeCombo()

    cboDocType.Clear

    Dim ws As Worksheet
    On Error Resume Next
    Set ws = GetDataSheet("DOCUMENT_TRACKER")
    On Error GoTo 0
    If ws Is Nothing Then Exit Sub

    Dim lastRow As Long
    lastRow = ws.Cells(ws.rows.count, "A").End(xlUp).Row
    If lastRow < 2 Then Exit Sub

    Dim dictTypes As Object
    Set dictTypes = CreateObject("Scripting.Dictionary")
    dictTypes.CompareMode = 1  ' case-insensitive

    Dim i As Long
    For i = 2 To lastRow
        Dim st As String
        st = UCase(Trim(CStr(ws.Cells(i, "O").value)))
        Dim passStatus As Boolean
        If mShowCompleted Then
            passStatus = (st = "COMPLETED" Or st = "ACCEPTED")
        Else
            passStatus = (st = "CHECKING" Or st = "PROCESSING")
        End If
        If passStatus Then
            Dim docType As String
            docType = Trim(CStr(ws.Cells(i, "C").value))
            If docType <> "" Then
                ' Bo qua IPC vi da co option rieng
                If UCase(docType) <> UCase("Interim Payment Certificate") Then
                    If Not dictTypes.Exists(docType) Then
                        dictTypes(docType) = 1
                    End If
                End If
            End If
        End If
    Next i

    Dim k As Variant
    For Each k In dictTypes.keys
        cboDocType.AddItem CStr(k)
    Next k

    If cboDocType.ListCount > 0 Then
        cboDocType.ListIndex = 0
    End If

End Sub

'================================================================
Public Sub LoadPendingList()

    Set mListWorkbook = Nothing
    If Not mDeleteOrphan Is Nothing Then mDeleteOrphan.Enabled = False

    lstPending.Clear
    ClearDetailPanel
    btnGoTo.Enabled = False

    Dim ws As Worksheet
    On Error Resume Next
    Set ws = GetDataSheet("DOCUMENT_TRACKER")
    On Error GoTo 0
    If ws Is Nothing Then
        lblCount.Caption = "Sheet DOCUMENT_TRACKER not found!"
        Exit Sub
    End If

    Dim lastRow As Long
    lastRow = ws.Cells(ws.rows.count, "A").End(xlUp).Row
    If lastRow < 2 Then
        lblCount.Caption = "No documents found."
        Exit Sub
    End If

    Set mListWorkbook = ws.Parent

    ' Xac dinh filter doc type theo mode
    Dim filterDocType As String
    filterDocType = ""
    Select Case mFilterMode
        Case 1
            filterDocType = "IPC"
        Case 2
            If cboDocType.ListIndex >= 0 Then
                filterDocType = CStr(cboDocType.value)
            Else
                lblCount.Caption = "Please select a document type."
                Exit Sub
            End If
    End Select

    Dim buf() As Variant
    ReDim buf(0 To lastRow - 2, 0 To 9)
    Dim cnt As Long: cnt = 0

    Dim i As Long
    For i = 2 To lastRow
        Dim st As String
        st = UCase(Trim(CStr(ws.Cells(i, "O").value)))
        Dim passStatus As Boolean
        If mShowCompleted Then
            passStatus = (st = "COMPLETED" Or st = "ACCEPTED")
        Else
            passStatus = (st = "CHECKING" Or st = "PROCESSING")
        End If
        If passStatus Then

            ' Kiem tra filter doc type
            Dim rowDocType As String
            rowDocType = Trim(CStr(ws.Cells(i, "C").value))

            Dim passFilter As Boolean
            passFilter = False
            Select Case mFilterMode
                Case 0   ' All Documents
                    passFilter = True
                Case 1   ' IPC only
                    passFilter = (UCase(rowDocType) = UCase(filterDocType))
                Case 2   ' Other - match selected combobox
                    passFilter = (UCase(rowDocType) = UCase(filterDocType))
            End Select

            If passFilter Then

                ' ---- Tim kiem theo tu khoa (giong txtKeyword) ----
                If mKeyword <> "" Then
                    Dim kw As String
                    kw = LCase(VN_NoSign(mKeyword))
                    Dim kwPartyB  As String: kwPartyB = ""
                    Dim kwConID   As String: kwConID = ""
                    Dim kwDocType As String: kwDocType = ""
                    Dim kwDocName As String: kwDocName = ""
                    Dim kwDocNo   As String: kwDocNo = ""
                    Dim kwConNo   As String: kwConNo = ""
                    Dim tmpCID As String: tmpCID = Trim(CStr(ws.Cells(i, "B").value))
                    kwConID = LCase(VN_NoSign(tmpCID))
                    kwDocType = LCase(VN_NoSign(rowDocType))
                    kwDocName = LCase(VN_NoSign(Trim(CStr(ws.Cells(i, "E").value))))
                    kwDocNo = LCase(VN_NoSign(Trim(CStr(ws.Cells(i, "D").value))))
                    If Not gDictContract Is Nothing Then
                        If gDictContract.Exists(tmpCID) Then
                            Dim kwR As Long: kwR = gDictContract(tmpCID)
                            kwPartyB = LCase(VN_NoSign(Trim(CStr(gArrContract(kwR, 8)))))
                            kwConNo = LCase(VN_NoSign(Trim(CStr(gArrContract(kwR, 2)))))
                        End If
                    End If
                    Dim kwPass As Boolean
                    kwPass = (InStr(kwPartyB, kw) > 0 Or InStr(kwConID, kw) > 0 _
                             Or InStr(kwConNo, kw) > 0 Or InStr(kwDocType, kw) > 0 _
                             Or InStr(kwDocName, kw) > 0 Or InStr(kwDocNo, kw) > 0)
                    If Not kwPass Then GoTo NEXT_ROW
                End If

                Dim docID As String
                docID = Trim(CStr(ws.Cells(i, "A").value))
                Dim contractID As String
                contractID = Trim(CStr(ws.Cells(i, "B").value))

                Dim grp As String
                If Left$(contractID, 2) = "A-" Then
                    grp = "A"
                ElseIf Left$(contractID, 2) = "B-" Then
                    grp = "B"
                Else
                    grp = "B"
                End If

                Dim contractNo As String: contractNo = contractID
                Dim partyB As String:     partyB = ""
                If Not gDictContract Is Nothing Then
                    If gDictContract.Exists(contractID) Then
                        Dim r As Long
                        r = gDictContract(contractID)
                        contractNo = Trim(CStr(gArrContract(r, 2)))
                        partyB = Trim(CStr(gArrContract(r, 8)))
                    End If
                End If

                Dim ngayCol As String
                If IsDate(ws.Cells(i, "I").value) Then
                    ngayCol = Format(ws.Cells(i, "I").value, "dd/mm/yy")
                Else
                    ngayCol = Trim(CStr(ws.Cells(i, "I").value))
                End If

                If Not ReminderContractExists(mListWorkbook, contractID) Then
                    partyB = "[Missing contract] " & contractID
                End If

                buf(cnt, 0) = contractNo
                buf(cnt, 1) = partyB
                buf(cnt, 2) = rowDocType
                buf(cnt, 3) = ExtractDocShortCode(Trim(CStr(ws.Cells(i, "D").value)))
                buf(cnt, 4) = ngayCol
                buf(cnt, 5) = Trim(CStr(ws.Cells(i, "O").value))
                buf(cnt, 6) = contractID
                buf(cnt, 7) = docID
                buf(cnt, 8) = grp
                ' col 9: NgayTra (col J) luu raw de sort completed
                If IsDate(ws.Cells(i, "J").value) Then
                    buf(cnt, 9) = ws.Cells(i, "J").value
                Else
                    buf(cnt, 9) = 0
                End If
                cnt = cnt + 1
            End If
        End If
NEXT_ROW:
    Next i

    If cnt = 0 Then
        Dim noMsg As String
        If mShowCompleted Then
            Select Case mFilterMode
                Case 0: noMsg = "No completed documents."
                Case 1: noMsg = "No completed IPC documents."
                Case 2: noMsg = "No completed documents of type: " & filterDocType
            End Select
        Else
            Select Case mFilterMode
                Case 0: noMsg = "No pending documents."
                Case 1: noMsg = "No pending IPC documents."
                Case 2: noMsg = "No pending documents of type: " & filterDocType
            End Select
        End If
        lblCount.Caption = noMsg
        Exit Sub
    End If

    ' Sap xep:
    '   Pending   -> group A truoc B, trong group ngay tang dan (NgayNhan)
    '   Completed -> moi nhat truoc (NgayTra giam dan, col J = buf col 9)
    Dim r1 As Long, r2 As Long, c As Long
    Dim tmpRow(0 To 9) As Variant
    For r1 = 0 To cnt - 2
        For r2 = r1 + 1 To cnt - 1
            Dim doSwap As Boolean: doSwap = False
            If mShowCompleted Then
                ' Sort completed: NgayTra giam dan (moi nhat len dau)
                Dim dt1 As Double: dt1 = 0
                Dim dt2 As Double: dt2 = 0
                On Error Resume Next
                If buf(r1, 9) <> 0 Then dt1 = CDbl(buf(r1, 9))
                If buf(r2, 9) <> 0 Then dt2 = CDbl(buf(r2, 9))
                On Error GoTo 0
                If dt1 < dt2 Then doSwap = True   ' giam dan: r2 moi hon r1 -> doi cho
            Else
                ' Sort pending: group A truoc B, trong group ngay tang dan
                Dim g1 As String: g1 = CStr(buf(r1, 8))
                Dim g2 As String: g2 = CStr(buf(r2, 8))
                If g1 > g2 Then
                    doSwap = True
                ElseIf g1 = g2 Then
                    Dim d1 As Date, d2 As Date
                    Dim ok1 As Boolean, ok2 As Boolean
                    ok1 = False: ok2 = False
                    On Error Resume Next
                    If buf(r1, 4) <> "" Then d1 = CDate(buf(r1, 4)): ok1 = (Err.Number = 0)
                    Err.Clear
                    If buf(r2, 4) <> "" Then d2 = CDate(buf(r2, 4)): ok2 = (Err.Number = 0)
                    Err.Clear
                    On Error GoTo 0
                    If ok1 And ok2 Then
                        If d1 > d2 Then doSwap = True
                    ElseIf Not ok1 And ok2 Then
                        doSwap = True
                    End If
                End If
            End If
            If doSwap Then
                For c = 0 To 9
                    tmpRow(c) = buf(r1, c)
                    buf(r1, c) = buf(r2, c)
                    buf(r2, c) = tmpRow(c)
                Next c
            End If
        Next r2
    Next r1

    ' Xay dung finalArr va mGroupArr
    Dim finalArr() As Variant
    ReDim finalArr(0 To cnt - 1, 0 To 7)
    ReDim mGroupArr(0 To cnt - 1)

    Dim rr As Long, cc As Long
    For rr = 0 To cnt - 1
        For cc = 0 To 7
            finalArr(rr, cc) = buf(rr, cc)
        Next cc
        mGroupArr(rr) = buf(rr, 8)
    Next rr

    On Error Resume Next
    lstPending.List = finalArr
    On Error GoTo 0

    ' Hien thi caption phu hop voi filter
    Dim filterLabel As String
    Select Case mFilterMode
        Case 0: filterLabel = "All"
        Case 1: filterLabel = "IPC"
        Case 2: filterLabel = filterDocType
    End Select
    Dim modeLabel As String
    If mShowCompleted Then modeLabel = "Completed" Else modeLabel = "Pending"
    lblCount.Caption = modeLabel & " [" & filterLabel & "]: " & cnt & " document(s)"

    If lstPending.ListCount > 0 Then
        lstPending.ListIndex = 0
        lstPending_Click
    End If

End Sub

'================================================================
Private Sub lstPending_Click()

    UpdateReminderActions

    Dim idx As Long
    idx = lstPending.ListIndex
    If idx < 0 Then
        ClearDetailPanel
        btnGoTo.Enabled = False
        Exit Sub
    End If

    ' UpdateReminderActions validates the current document and contract.

    Dim contractID As String
    Dim docID As String
    contractID = CStr(lstPending.List(idx, 6))
    docID = CStr(lstPending.List(idx, 7))

    Dim grp As String
    If idx <= UBound(mGroupArr) Then grp = mGroupArr(idx) Else grp = "B"

    Dim dateLabel As String
    If grp = "A" Then dateLabel = "Processed Date" Else dateLabel = "Received Date"

    Dim contractNo As String: contractNo = ""
    Dim partyB As String:     partyB = ""
    Dim scope As String:      scope = ""
    If Not gDictContract Is Nothing Then
        If gDictContract.Exists(contractID) Then
            Dim r As Long
            r = gDictContract(contractID)
            contractNo = Trim(CStr(gArrContract(r, 2)))
            partyB = Trim(CStr(gArrContract(r, 8)))
            scope = Trim(CStr(gArrContract(r, 11)))
        End If
    End If

    Dim ws As Worksheet
    On Error Resume Next
    Set ws = GetDataSheet("DOCUMENT_TRACKER")
    On Error GoTo 0
    If ws Is Nothing Then Exit Sub

    Dim lastRow As Long
    lastRow = ws.Cells(ws.rows.count, "A").End(xlUp).Row

    Dim i As Long
    Dim foundRow As Long: foundRow = -1
    For i = 2 To lastRow
        If Trim(CStr(ws.Cells(i, "A").value)) = docID Then
            foundRow = i
            Exit For
        End If
    Next i

    DrawDetailPanel grp, dateLabel, contractID, partyB, scope, ws, foundRow

End Sub

'================================================================
' ClearDetailPanel — Xoa toan bo Label dong (DYN_DT_*) trong fraDetail
'================================================================
Private Sub ClearDetailPanel()
    On Error Resume Next
    Dim toDel() As String
    ReDim toDel(0 To fraDetail.Controls.count)
    Dim n As Long: n = 0
    Dim ctrl As MSForms.Control
    For Each ctrl In fraDetail.Controls
        If Left$(ctrl.Name, 7) = "DYN_DT_" Then
            toDel(n) = ctrl.Name
            n = n + 1
        End If
    Next ctrl
    Dim j As Long
    For j = 0 To n - 1
        fraDetail.Controls.Remove toDel(j)
    Next j
    On Error GoTo 0
End Sub

'================================================================
' DrawDetailPanel — Ve chi tiet ho so bang Label dong (khong scrollbar)
'
' grp       : "A" hoac "B" (quyet dinh mau header)
' dateLabel : "Processed Date" hoac "Received Date"
' contractNo, partyB, scope : tu gArrContract
' ws        : sheet DOCUMENT_TRACKER
' wsRow     : dong du lieu (-1 neu khong tim thay)
'================================================================
Private Sub DrawDetailPanel( _
    ByVal grp As String, _
    ByVal dateLabel As String, _
    ByVal contractID As String, _
    ByVal partyB As String, _
    ByVal scope As String, _
    ByVal ws As Worksheet, _
    ByVal wsRow As Long)

    On Error GoTo ERR_DRAW
    ClearDetailPanel

    '---- Layout constants ----
    Const ROW_H  As Single = 15   ' chieu cao moi dong du lieu
    Const HDR_H  As Single = 18   ' chieu cao header
    Const SEP_H  As Single = 5    ' duong ke ngang
    Const W_KEY  As Single = 110  ' do rong cot nhan (field name)
    Const PAD_L  As Single = 1    ' le trai
    Const FONT_SZ As Single = 8

    Dim frm As MSForms.Frame
    Set frm = fraDetail

    Dim fW As Single
    fW = frm.InsideWidth - 2
    If fW < 60 Then fW = 60

    '---- Mau theo group ----
    Dim clrHdr   As Long
    Dim clrHdrFG As Long
    Dim clrKey1  As Long
    Dim clrKey2  As Long
    Dim clrVal1  As Long
    Dim clrVal2  As Long
    Dim clrKeyFG As Long
    Dim clrValFG As Long
    Dim clrSep   As Long

    If grp = "A" Then
        clrHdr = RGB(31, 73, 125)
        clrHdrFG = RGB(255, 255, 255)
        clrKey1 = RGB(210, 228, 255)
        clrKey2 = RGB(200, 218, 248)
        clrVal1 = RGB(235, 244, 255)
        clrVal2 = RGB(225, 237, 255)
        clrKeyFG = RGB(31, 56, 100)
        clrValFG = RGB(20, 40, 80)
        clrSep = RGB(31, 73, 125)
    Else
        clrHdr = RGB(120, 40, 40)
        clrHdrFG = RGB(255, 255, 255)
        clrKey1 = RGB(255, 225, 225)
        clrKey2 = RGB(248, 212, 212)
        clrVal1 = RGB(255, 242, 242)
        clrVal2 = RGB(255, 232, 232)
        clrKeyFG = RGB(100, 31, 31)
        clrValFG = RGB(80, 20, 20)
        clrSep = RGB(120, 40, 40)
    End If

    Dim Y As Single: Y = 1
    Dim rowIdx As Long: rowIdx = 0

    '================================================================
    ' Helper: them 1 dong (key + value)
    ' Dung bien rowIdx de xen ke mau nen
    '================================================================

    '---- HEADER ----
    Dim lHdr As MSForms.Label
    Set lHdr = frm.Controls.Add("Forms.Label.1", "DYN_DT_HDR")
    With lHdr
        .Caption = "  " & IIf(grp = "A", "GROUP A", "GROUP B") & _
                   "  |  " & BuildDisplayContractNo(contractID)
        .Left = PAD_L: .Top = Y: .Width = fW: .Height = HDR_H
        .Font.Size = FONT_SZ + 1: .Font.Bold = True
        .ForeColor = clrHdrFG: .BackColor = clrHdr: .BackStyle = 1
        .WordWrap = False
    End With
    Y = Y + HDR_H + 1

    '----------------------------------------------------------------
    ' Macro noi bo: them 1 dong label key + value
    '----------------------------------------------------------------
    Dim clrK As Long, clrV As Long

    ' --- Party B ---
    clrK = IIf(rowIdx Mod 2 = 0, clrKey1, clrKey2)
    clrV = IIf(rowIdx Mod 2 = 0, clrVal1, clrVal2)
    Dim lK0 As MSForms.Label
    Set lK0 = frm.Controls.Add("Forms.Label.1", "DYN_DT_K0")
    With lK0
        .Caption = "  Party B"
        .Left = PAD_L: .Top = Y: .Width = W_KEY: .Height = ROW_H
        .Font.Size = FONT_SZ: .Font.Bold = True
        .ForeColor = clrKeyFG: .BackColor = clrK: .BackStyle = 1: .WordWrap = False
    End With
    Dim lV0 As MSForms.TextBox
    Set lV0 = frm.Controls.Add("Forms.TextBox.1", "DYN_DT_V0")
    With lV0
        .Text = partyB
        .Left = PAD_L + W_KEY: .Top = Y: .Width = fW - W_KEY: .Height = ROW_H
        .Font.Size = FONT_SZ: .Font.Bold = False
        .ForeColor = clrValFG: .BackColor = clrV
        .Locked = True: .BorderStyle = 1: .SpecialEffect = 0: .BorderColor = RGB(189, 215, 238): .ScrollBars = 0
        .MultiLine = False: .WordWrap = False
        .EnterKeyBehavior = False: .TabKeyBehavior = False
    End With
    Y = Y + ROW_H: rowIdx = rowIdx + 1

    ' --- Scope (2 dong chieu cao de bao gom text dai) ---
    clrK = IIf(rowIdx Mod 2 = 0, clrKey1, clrKey2)
    clrV = IIf(rowIdx Mod 2 = 0, clrVal1, clrVal2)
    Dim lKSc As MSForms.Label
    Set lKSc = frm.Controls.Add("Forms.Label.1", "DYN_DT_KSC")
    With lKSc
        .Caption = "  Scope"
        .Left = PAD_L: .Top = Y: .Width = W_KEY: .Height = ROW_H * 2
        .Font.Size = FONT_SZ: .Font.Bold = True
        .ForeColor = clrKeyFG: .BackColor = clrK: .BackStyle = 1: .WordWrap = False
    End With
    Dim lVSc As MSForms.TextBox
    Set lVSc = frm.Controls.Add("Forms.TextBox.1", "DYN_DT_VSC")
    With lVSc
        .Text = scope
        .Left = PAD_L + W_KEY: .Top = Y: .Width = fW - W_KEY: .Height = ROW_H * 2
        .Font.Size = FONT_SZ: .Font.Bold = False
        .ForeColor = clrValFG: .BackColor = clrV
        .Locked = True: .BorderStyle = 1: .SpecialEffect = 0: .BorderColor = RGB(189, 215, 238): .ScrollBars = 0
        .MultiLine = True: .WordWrap = True
        .EnterKeyBehavior = False: .TabKeyBehavior = False
    End With
    Y = Y + ROW_H * 2: rowIdx = rowIdx + 1

    ' --- Separator ---
    Dim lSep As MSForms.Label
    Set lSep = frm.Controls.Add("Forms.Label.1", "DYN_DT_SEP")
    With lSep
        .Caption = ""
        .Left = PAD_L: .Top = Y: .Width = fW: .Height = SEP_H
        .BackColor = clrSep: .BackStyle = 1
    End With
    Y = Y + SEP_H + 1

    '---- Doc fields: tu worksheet ----
    Dim fType As String:  fType = ""
    Dim fName As String:  fName = ""
    Dim fDocNo As String: fDocNo = ""
    Dim fStat As String:  fStat = ""
    Dim fNgay As String:  fNgay = ""
    Dim fSupp As String:  fSupp = ""
    Dim fVer As String:   fVer = ""
    Dim fNote As String:  fNote = ""
    Dim fLink As String:  fLink = ""

    If wsRow > 0 Then
        fType = Trim(CStr(ws.Cells(wsRow, "C").value))
        fName = Trim(CStr(ws.Cells(wsRow, "E").value))
        fDocNo = Trim(CStr(ws.Cells(wsRow, "D").value))
        fStat = Trim(CStr(ws.Cells(wsRow, "O").value))
        If IsDate(ws.Cells(wsRow, "I").value) Then
            fNgay = Format(ws.Cells(wsRow, "I").value, "dd/mm/yyyy")
        Else
            fNgay = Trim(CStr(ws.Cells(wsRow, "I").value))
        End If
        fSupp = Trim(CStr(ws.Cells(wsRow, "M").value))
        fVer = Trim(CStr(ws.Cells(wsRow, "N").value))
        fNote = Trim(CStr(ws.Cells(wsRow, "P").value))
        fLink = Trim(CStr(ws.Cells(wsRow, "F").value))
    End If

    ' Mau Status
    Dim clrStatBg As Long: clrStatBg = 0
    Dim clrStatFG As Long: clrStatFG = 0
    Select Case UCase(Trim(fStat))
        Case "CHECKING", "PROCESSING"

    fieldKeys(7) = "Note":               fieldVals(7) = fNote:  fieldH(7) = ROW_H * 2: fieldIsNote(7) = True
    fieldKeys(8) = "Link":               fieldVals(8) = fLink:  fieldH(8) = ROW_H

    For fi = 0 To 8
        clrK = IIf(rowIdx Mod 2 = 0, clrKey1, clrKey2)

        useStatColor = (fi = 3 And clrStatBg <> 0)

        Set lK = frm.Controls.Add("Forms.Label.1", "DYN_DT_K" & (fi + 1))

            .ForeColor = clrKeyFG

        End With

        Set lV = frm.Controls.Add("Forms.TextBox.1", "DYN_DT_V" & (fi + 1))

            .ForeColor = IIf(useStatColor, clrStatFG, clrValFG)
            .BackColor = IIf(useStatColor, clrStatBg, clrV)
            .Locked = True: .BorderStyle = 1: .SpecialEffect = 0: .BorderColor = RGB(189, 215, 238): .ScrollBars = 0

            .EnterKeyBehavior = False: .TabKeyBehavior = False
        End With

        Y = Y + fieldH(fi): rowIdx = rowIdx + 1
    Next fi

ERR_DRAW:

End Sub

Private Function PadRR(ByVal s As String, ByVal n As Integer) As String
    If Len(s) >= n Then PadRR = Left$(s, n) Else PadRR = s & String(n - Len(s), " ")
End Function

'================================================================
' Trich xuat phan so cuoi cua document number + version (A-Z) neu co
' Neu phan so chi 1 chu so (<10) -> pad thanh 2 chu so
' Vi du:
'   "260619.CHC1.MN.IPC.VEC.001"  -> "001"   (da du 3 chu so, giu nguyen)
'   "260619.CHC1.MN.IPC.VEC.001A" -> "001A"
'   "260619.CHC1.MN.IPC.VEC.3"    -> "03"    (1 chu so -> pad)
'   "260619.CHC1.MN.IPC.VEC.3A"   -> "03A"   (1 chu so + version -> pad)
'   "ABC-2026-001B"               -> "001B"
'   "001"                         -> "001"
'   ""                            -> ""
'================================================================
Private Function ExtractDocShortCode(ByVal docNum As String) As String

    If docNum = "" Then
        ExtractDocShortCode = ""
        Exit Function
    End If

    ' Lay phan sau dau "." cuoi cung
    Dim lastDot As Long
    lastDot = 0
    Dim p As Long
    For p = Len(docNum) To 1 Step -1
        If mID$(docNum, p, 1) = "." Then
            lastDot = p
            Exit For
        End If
    Next p

    Dim tail As String
    If lastDot > 0 Then
        tail = mID$(docNum, lastDot + 1)
    Else
        ' Khong co dau ".", thu tim dau "-" cuoi
        Dim lastDash As Long
        lastDash = 0
        For p = Len(docNum) To 1 Step -1
            If mID$(docNum, p, 1) = "-" Then
                lastDash = p
                Exit For
            End If
        Next p
        If lastDash > 0 Then
            tail = mID$(docNum, lastDash + 1)
        Else
            tail = docNum
        End If
    End If

    Dim tailTrim As String
    tailTrim = Trim$(tail)

    If tailTrim = "" Then
        ExtractDocShortCode = docNum   ' fallback
        Exit Function
    End If

    ' Kiem tra ky tu cuoi co phai version letter (A-Z) khong
    Dim lastChar As String
    lastChar = Right$(tailTrim, 1)
    Dim hasVersion As Boolean
    hasVersion = (lastChar >= "A" And lastChar <= "Z") Or _
                 (lastChar >= "a" And lastChar <= "z")

    Dim numPart As String
    Dim verPart As String
    If hasVersion Then
        numPart = Left$(tailTrim, Len(tailTrim) - 1)
        verPart = lastChar
    Else
        numPart = tailTrim
        verPart = ""
    End If

    ' Kiem tra numPart toan chu so
    Dim allDigits As Boolean
    allDigits = True
    Dim ch As String
    For p = 1 To Len(numPart)
        ch = mID$(numPart, p, 1)
        If ch < "0" Or ch > "9" Then
            allDigits = False
            Exit For
        End If
    Next p

    If allDigits And Len(numPart) > 0 Then
        ' Pad len 2 chu so neu chi co 1 chu so
        If Len(numPart) = 1 Then numPart = "0" & numPart
        ExtractDocShortCode = numPart & verPart
    Else
        ' Tail khong phai so -> fallback giu nguyen
        ExtractDocShortCode = docNum
    End If

End Function

Private Sub btnGoTo_Click()

    If Not CanNavigateReminderSelection() Then Exit Sub

    Dim idx As Long
    idx = lstPending.ListIndex
    If idx < 0 Then Exit Sub

    Dim contractID As String
    Dim docID As String
    Dim docType As String
    contractID = CStr(lstPending.List(idx, 6))
    docID = CStr(lstPending.List(idx, 7))
    docType = CStr(lstPending.List(idx, 2))   ' cot C: kieu tai lieu

    If Not gContractLoaded Then BuildGlobalContractCache
    If Not gPaymentLoaded Then BuildPaymentCache

    If Not IsFormAlive(mContractSearch) Then
        Set mContractSearch = Nothing
    End If

    Dim needLoadList As Boolean: needLoadList = False

    If mContractSearch Is Nothing Then
        Dim newCS As frmContractSearch
        Set newCS = New frmContractSearch
        Set mContractSearch = newCS
        newCS.SetReminderRef Me
        needLoadList = True
    End If

    mContractSearch.Show vbModeless

    If needLoadList Then
        mContractSearch.LoadContractList
    End If

    On Error Resume Next
    mContractSearch.NavigateToDocument contractID, docID, docType
    On Error GoTo 0

    Me.Hide

    If Not mHomePage Is Nothing Then
        If IsFormAlive(mHomePage) Then
            mHomePage.Hide
        Else
            Set mHomePage = Nothing
        End If
    End If

End Sub

Private Function ReminderContractExists(ByVal wb As Workbook, ByVal contractID As String) As Boolean
    Dim ws As Worksheet, r As Long, lastRow As Long
    Set ws = wb.Worksheets("CONTRACT_DB")
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    For r = 2 To lastRow
        If Len(Trim$(CStr(ws.Cells(r, "A").Value2))) > 0 Then
            If StrComp(Trim$(CStr(ws.Cells(r, "A").Value2)), Trim$(contractID), vbTextCompare) = 0 Then
                ReminderContractExists = True
                Exit Function
            End If
        End If
    Next r
End Function

Private Function ResolveReminderSelection(ByRef wb As Workbook, ByRef ws As Worksheet, _
        ByRef rowNo As Long, ByRef docID As String, ByRef contractID As String) As Boolean
    Dim i As Long, matches As Long, idx As Long
    idx = lstPending.ListIndex
    If idx < 0 Then Exit Function
    If mListWorkbook Is Nothing Then Exit Function
    Set wb = GetDataWorkbook()
    If wb Is Nothing Then Exit Function
    If Not (wb Is mListWorkbook) Then Exit Function
    Set ws = wb.Worksheets("DOCUMENT_TRACKER")
    docID = Trim$(CStr(lstPending.List(idx, 7)))
    contractID = Trim$(CStr(lstPending.List(idx, 6)))
    If Len(docID) = 0 Then Exit Function
    For i = 2 To ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
        If StrComp(Trim$(CStr(ws.Cells(i, "A").Value2)), docID, vbTextCompare) = 0 Then
            matches = matches + 1
            rowNo = i
        End If
    Next i
    If matches <> 1 Then Exit Function
    If StrComp(Trim$(CStr(ws.Cells(rowNo, "B").Value2)), contractID, vbTextCompare) <> 0 Then Exit Function
    ResolveReminderSelection = True
End Function

Private Sub UpdateReminderActions()
    Dim wb As Workbook, ws As Worksheet, rowNo As Long
    Dim docID As String, contractID As String, exists As Boolean
    btnGoTo.Enabled = False
    If Not mDeleteOrphan Is Nothing Then mDeleteOrphan.Enabled = False
    On Error GoTo Done
    If Not ResolveReminderSelection(wb, ws, rowNo, docID, contractID) Then Exit Sub
    exists = ReminderContractExists(wb, contractID)
    btnGoTo.Enabled = exists
    If Not mDeleteOrphan Is Nothing Then
        mDeleteOrphan.Enabled = Not exists And Not wb.ReadOnly And Not ws.ProtectContents
    End If
Done:
End Sub

Private Function CanNavigateReminderSelection() As Boolean
    Dim wb As Workbook, ws As Worksheet, rowNo As Long
    Dim docID As String, contractID As String
    On Error GoTo Unavailable
    If Not ResolveReminderSelection(wb, ws, rowNo, docID, contractID) Then GoTo Unavailable
    If Not ReminderContractExists(wb, contractID) Then
        MsgBox "This document has no existing contract. Use Delete orphan to remove it.", vbInformation, "Reminders"
        UpdateReminderActions
        Exit Function
    End If
    CanNavigateReminderSelection = True
    Exit Function
Unavailable:
    MsgBox "The selected document or data project has changed. Click Refresh and try again.", vbExclamation, "Reminders"
    UpdateReminderActions
End Function

Private Sub mDeleteOrphan_Click()
    Dim wb As Workbook, ws As Worksheet, rowNo As Long
    Dim docID As String, contractID As String, docNo As String
    Dim confirmedID As String, confirmedCID As String, confirmedNo As String
    Dim confirmedWorkbook As Workbook, deleted As Boolean, saved As Boolean
    Dim errorText As String
    On Error GoTo Failed
    If Not ResolveReminderSelection(wb, ws, rowNo, docID, contractID) Then GoTo Changed
    If ReminderContractExists(wb, contractID) Then
        MsgBox "This document still belongs to an existing contract and cannot be deleted here.", vbExclamation, "Reminders"
        UpdateReminderActions
        Exit Sub
    End If
    If wb.ReadOnly Or ws.ProtectContents Then
        MsgBox "The data file is read-only or DOCUMENT_TRACKER is protected.", vbExclamation, "Reminders"
        Exit Sub
    End If
    docNo = CStr(ws.Cells(rowNo, "D").Value2)
    confirmedID = docID
    confirmedCID = contractID
    confirmedNo = docNo
    Set confirmedWorkbook = wb
    If MsgBox("Delete this orphan document from the data file?" & vbCrLf & vbCrLf & _
            "DocumentID: " & docID & vbCrLf & "Document: " & docNo & vbCrLf & _
            "Missing ContractID: " & contractID & vbCrLf & vbCrLf & _
            "The linked file will NOT be deleted." & vbCrLf & _
            "Saving also saves other pending changes in this data file.", _
            vbYesNo + vbQuestion + vbDefaultButton2, "Delete orphan document") <> vbYes Then Exit Sub

    ' Re-resolve the stable ID after confirmation; never trust a cached row number.
    If Not ResolveReminderSelection(wb, ws, rowNo, docID, contractID) Then GoTo Changed
    If Not (wb Is confirmedWorkbook) Then GoTo Changed
    If docID <> confirmedID Or contractID <> confirmedCID Then GoTo Changed
    If CStr(ws.Cells(rowNo, "D").Value2) <> confirmedNo Then GoTo Changed
    If ReminderContractExists(wb, contractID) Then GoTo Changed
    If wb.ReadOnly Or ws.ProtectContents Then GoTo Changed
    ws.Rows(rowNo).Delete
    deleted = True
    gDocLoaded = False
    wb.Save
    If Not wb.Saved Then Err.Raise vbObjectError + 2101, "frmReminder", "The data workbook was not saved."
    saved = True
    BuildDocTrackerCache
    BuildGlobalContractCache
    LoadPendingList
    MsgBox "Orphan document deleted and data file saved.", vbInformation, "Reminders"
    Exit Sub
Changed:
    MsgBox "The selected document or its contract has changed. Click Refresh and try again.", vbExclamation, "Reminders"
    UpdateReminderActions
    Exit Sub
Failed:
    errorText = Err.Description
    On Error Resume Next
    If deleted Then
        gDocLoaded = False
        BuildDocTrackerCache
        LoadPendingList
    End If
    UpdateReminderActions
    On Error GoTo 0
    If saved Then
        MsgBox "The document was deleted and saved, but the display could not be refreshed: " & errorText, vbExclamation, "Reminders"
    ElseIf deleted Then
        MsgBox "The document was removed in memory, but saving failed. The deletion is not confirmed on disk. " & _
            "Keep the data workbook open and resolve the save error: " & errorText, vbCritical, "Reminders"
    Else
        MsgBox "The document was not deleted: " & errorText, vbExclamation, "Reminders"
    End If
End Sub
