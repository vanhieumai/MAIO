' ============================================================
' CONVERT IMAGE TO BASE64
' ============================================================
Private Function ImageToBase64(filePath As String) As String
    On Error GoTo EH

    Dim stream As Object
    Set stream = CreateObject("ADODB.Stream")
    stream.Type = 1
    stream.Open
    stream.LoadFromFile filePath

    Dim bytes() As Byte
    bytes = stream.Read
    stream.Close

    Dim xml As Object
    Set xml = CreateObject("MSXML2.DOMDocument")
    Dim node As Object
    Set node = xml.createElement("b64")
    node.DataType = "bin.base64"
    node.nodeTypedValue = bytes
    ImageToBase64 = Replace(Replace(node.Text, vbLf, ""), vbCr, "")
    Exit Function
EH:
    ImageToBase64 = ""
End Function

' ============================================================
' GET / SAVE APP SETTING
' ============================================================
Public Function GetAppSetting(settingKey As String) As String
    Dim ws As Worksheet
    Set ws = GetSettingsSheet()
    If ws Is Nothing Then Exit Function
    Dim lastRow As Long
    lastRow = ws.Cells(ws.rows.count, "A").End(xlUp).Row
    Dim i As Long
    For i = 2 To lastRow
        If Trim(CStr(ws.Cells(i, 1).value)) = settingKey Then
            GetAppSetting = Trim(CStr(ws.Cells(i, 2).value))
            Exit Function
        End If
    Next i
End Function

Public Sub SaveAppSetting(settingKey As String, settingValue As String)
    Dim ws As Worksheet
    Set ws = GetSettingsSheet()
    If ws Is Nothing Then Exit Sub
    Dim lastRow As Long
    lastRow = ws.Cells(ws.rows.count, "A").End(xlUp).Row
    Dim i As Long
    For i = 2 To lastRow
        If Trim(CStr(ws.Cells(i, 1).value)) = settingKey Then
            ws.Cells(i, 2).value = settingValue
            SaveAIFile
            Exit Sub
        End If
    Next i
    ws.Cells(lastRow + 1, 1).value = settingKey
    ws.Cells(lastRow + 1, 2).value = settingValue
    SaveAIFile
End Sub

Public Sub SaveApiKeyToHistory(newKey As String)
    newKey = Trim(newKey)
    If newKey = "" Then Exit Sub
    Dim ws As Worksheet
    Set ws = GetSettingsSheet()
    If ws Is Nothing Then Exit Sub
    Dim lastRow As Long: lastRow = ws.Cells(ws.rows.count, "A").End(xlUp).Row
    Dim maxIdx As Long: maxIdx = 0
    Dim i As Long
    For i = 2 To lastRow
        Dim cellKey As String
        cellKey = Trim(CStr(ws.Cells(i, 1).value))
        If Left(cellKey, Len(APIKEY_PREFIX)) = APIKEY_PREFIX Then
            If Trim(CStr(ws.Cells(i, 2).value)) = newKey Then Exit Sub
            Dim idx As Long
            On Error Resume Next
            idx = CLng(mID(cellKey, Len(APIKEY_PREFIX) + 1))
            On Error GoTo 0
            If idx > maxIdx Then maxIdx = idx
        End If
    Next i
    ws.Cells(lastRow + 1, 1).value = APIKEY_PREFIX & (maxIdx + 1)
    ws.Cells(lastRow + 1, 2).value = newKey
    SaveAIFile
End Sub

Private Sub UpdateProviderUI()
    If mBusy Then Exit Sub
    On Error GoTo Failed
    mLoadingModels = True
    Dim provider As String, models() As String, savedModel As String
    Dim i As Long, cached As String, parts As Variant, entry As Variant
    provider = DetectProvider(GetApiKey())
    lblProviderName.Caption = GetProviderDisplayName(provider)
    If GetApiKey() = "" Then
        lblKeyValue.Caption = "Click to enter API Key"
    Else
        lblKeyValue.Caption = MaskKey(GetApiKey())
    End If
    mFree.Visible = (provider = "OPENROUTER")
    mRefresh.Enabled = (provider = "GROQ" Or provider = "OPENROUTER")
    Set mVision = CreateObject("Scripting.Dictionary")
    savedModel = LoadSelectedModel()
    cboModel.Clear
    cached = GetAppSetting(ModelCacheKey(provider))
    If cached <> "" Then
        parts = Split(cached, "|")
        For Each entry In parts
            If Len(CStr(entry)) > 2 Then
                cboModel.AddItem Mid$(CStr(entry), 3)
                mVision(Mid$(CStr(entry), 3)) = (Left$(CStr(entry), 1) = "1")
            End If
        Next entry
    Else
        models = GetProviderModels(provider)
        For i = LBound(models) To UBound(models)
            cboModel.AddItem models(i)
        Next i
    End If
    SelectModel savedModel
    mLoadingModels = False
    UpdateVisionButton
    Exit Sub
Failed:
    mLoadingModels = False
    MsgBox "Model list: " & Err.Description, vbExclamation
End Sub
Private Sub cboModel_Change()
    If mLoadingModels Or mBusy Then Exit Sub
    If cboModel.ListIndex >= 0 Then SaveSelectedModel CStr(cboModel.Value)
    UpdateVisionButton
End Sub

' CLICK KEY LABEL TO MANAGE KEYS
' ============================================================
Private Sub lblKeyValue_Click()
    If mBusy Then Exit Sub
    Dim savedKeys() As String
    savedKeys = GetSavedApiKeys()

    Dim hasSaved As Boolean
    hasSaved = False
    If UBound(savedKeys) >= 0 Then
        If savedKeys(0) <> "" Then hasSaved = True
    End If

    Dim menu As String
    Dim i As Integer

    If hasSaved Then
        menu = "=== SAVED KEYS ===" & vbCrLf
        For i = 0 To UBound(savedKeys)
            If savedKeys(i) <> "" Then
                Dim provider As String
                provider = DetectProvider(savedKeys(i))
                menu = menu & "  " & (i + 1) & ". [" & GetProviderDisplayName(provider) & "] " & MaskKey(savedKeys(i)) & vbCrLf
            End If
        Next i
        menu = menu & vbCrLf
        menu = menu & "Enter number (1-" & (UBound(savedKeys) + 1) & ") to select key." & vbCrLf
        menu = menu & "Enter N to add new key." & vbCrLf
        menu = menu & "Enter D1, D2... to delete key."
    Else
        menu = "No keys saved." & vbCrLf & "Enter N to add new key."
    End If

    Dim choice As String
    choice = Trim(InputBox(menu, "Manage API Key"))
    If choice = "" Then Exit Sub

    If UCase(choice) = "N" Then
        Dim newKey As String
        newKey = Trim(InputBox( _
            "Enter new API Key:" & vbCrLf & vbCrLf & _
            "  gsk_...    -> Groq" & vbCrLf & _
            "  nvapi-...  -> NVIDIA NIM" & vbCrLf & _
            "  sk-or-...  -> OpenRouter" & vbCrLf & _
            "  AIza...    -> Google Gemini" & vbCrLf & _
            "  sk-...     -> OpenAI", _
            "Add New API Key"))
        If newKey = "" Then Exit Sub
        SetApiKey newKey
        SaveApiKeyToHistory newKey
        If IsHexKey32(newKey) Then
            Dim azureEp As String
            azureEp = Trim(InputBox( _
                "Azure key detected (32 hex chars)." & vbCrLf & vbCrLf & _
                "Enter Azure endpoint URL:", _
                "Azure Endpoint", GetAzureEndpoint()))
            If azureEp <> "" Then SaveAzureEndpoint azureEp
        End If

    ElseIf UCase(Left(choice, 1)) = "D" Then
        Dim delIdxStr As String
        delIdxStr = mID(choice, 2)
        Dim delIdx As Long
        On Error Resume Next
        delIdx = -1
        delIdx = CLng(delIdxStr) - 1
        On Error GoTo 0
        If delIdx >= 0 And delIdx <= UBound(savedKeys)  Then
            Dim delKey As String
            delKey = savedKeys(delIdx)
            If GetApiKey() = delKey Then SetApiKey ""
            DeleteApiKeyFromHistory delKey
            MsgBox "Key deleted: " & MaskKey(delKey), vbInformation
        Else
            MsgBox "Invalid number.", vbExclamation
        End If

    Else
        Dim selIdx As Long
        On Error Resume Next
        selIdx = -1
        selIdx = CLng(choice) - 1
        On Error GoTo 0
        If selIdx >= 0 And selIdx <= UBound(savedKeys)  Then
            SetApiKey savedKeys(selIdx)
        Else
            MsgBox "Invalid number.", vbExclamation
            Exit Sub
        End If
    End If

    UpdateProviderUI
    lblRespone.value = ""
End Sub

' HISTORY BUTTON
' ============================================================
Private Sub btnHistory_Click()
    If mBusy Then Exit Sub
    Dim ws As Worksheet
    On Error Resume Next
    Set ws = GetAISheet("AI_NOTES")
    On Error GoTo 0

    If ws Is Nothing Then
        MsgBox "No history saved.", vbInformation
        Exit Sub
    End If

    Dim lastRow As Long
    lastRow = ws.Cells(ws.rows.count, 1).End(xlUp).Row
    If lastRow < 2 Then
        MsgBox "No history saved.", vbInformation
        Exit Sub
    End If

    Dim startRow As Long
    startRow = IIf(lastRow > 51, lastRow - 49, 2)

    Dim menu As String
    menu = "=== SAVED QUESTION HISTORY ===" & vbCrLf & vbCrLf

    Dim entries() As Long
    Dim entryCount As Long
    ReDim entries(lastRow - startRow)

    Dim r As Long
    For r = lastRow To startRow Step -1
        Dim q As String
        q = Trim(CStr(ws.Cells(r, 2).value))
        If q <> "" Then
            entries(entryCount) = r
            entryCount = entryCount + 1
            Dim preview As String
            preview = Left(q, 60) & IIf(Len(q) > 60, "...", "")
            Dim timeStr As String
            On Error Resume Next
            timeStr = Format(ws.Cells(r, 1).value, "dd/mm hh:mm")
            On Error GoTo 0
            menu = menu & entryCount & ". [" & timeStr & "] " & preview & vbCrLf
        End If
    Next r

    If entryCount = 0 Then
        MsgBox "No history saved.", vbInformation
        Exit Sub
    End If

    menu = menu & vbCrLf & "Enter number to reload (1-" & entryCount & "):" & vbCrLf
    menu = menu & "Enter X to delete an entry (e.g. X3)."

    Dim choice As String
    choice = Trim(InputBox(menu, "Question History"))
    If choice = "" Then Exit Sub

    If UCase(Left(choice, 1)) = "X" Then
        Dim delNum As Long
        On Error Resume Next
        delNum = CLng(mID(choice, 2))
        On Error GoTo 0
        If delNum >= 1 And delNum <= entryCount Then
            Dim delRow As Long
            delRow = entries(delNum - 1)
            If MsgBox("Delete this entry?" & vbCrLf & Left(Trim(CStr(ws.Cells(delRow, 2).value)), 40), _
                      vbQuestion + vbYesNo) = vbYes Then
                ws.rows(delRow).Delete
                SaveAIFile
                MsgBox "Deleted.", vbInformation
            End If
        Else
            MsgBox "Invalid number.", vbExclamation
        End If
        Exit Sub
    End If

    Dim selNum As Long
    On Error Resume Next
    selNum = CLng(choice)
    On Error GoTo 0

    If selNum >= 1 And selNum <= entryCount Then
        Dim selRow As Long
        selRow = entries(selNum - 1)
        txtQuestion.Text = Trim(CStr(ws.Cells(selRow, 2).value))
        Dim loadedA As String
        loadedA = Trim(CStr(ws.Cells(selRow, 3).value))
        Dim dispProv As String
        Dim lProv As String: lProv = Trim(CStr(ws.Cells(selRow, 4).value))
        Dim lModel As String: lModel = Trim(CStr(ws.Cells(selRow, 5).value))
        If lProv <> "" And lModel <> "" Then
            dispProv = lProv & "  (" & lModel & ")"
        ElseIf lProv <> "" Then
            dispProv = lProv
        Else
            dispProv = "?"
        End If
        lblRespone.value = "AI: " & dispProv & vbCrLf & vbCrLf & loadedA
    Else
        MsgBox "Invalid number.", vbExclamation
    End If
End Sub

Public Sub DeleteApiKeyFromHistory(targetKey As String)
    targetKey = Trim(targetKey)
    Dim ws As Worksheet
    Set ws = GetSettingsSheet()
    If ws Is Nothing Then Exit Sub
    Dim lastRow As Long: lastRow = ws.Cells(ws.rows.count, "A").End(xlUp).Row
    Dim i As Long
    For i = lastRow To 2 Step -1
        Dim cellKey As String: cellKey = Trim(CStr(ws.Cells(i, 1).value))
        If Left(cellKey, Len(APIKEY_PREFIX)) = APIKEY_PREFIX Then
            If Trim(CStr(ws.Cells(i, 2).value)) = targetKey Then
                ws.rows(i).Delete
                SaveAIFile
                Exit For
            End If
        End If
    Next i
End Sub

Public Function GetSavedApiKeys() As String()
    Dim ws As Worksheet
    Set ws = GetSettingsSheet()

    Dim emptyKeys(0) As String
    emptyKeys(0) = ""

    If ws Is Nothing Then
        GetSavedApiKeys = emptyKeys
        Exit Function
    End If

    Dim lastRow As Long
    lastRow = ws.Cells(ws.rows.count, "A").End(xlUp).Row

    Dim count As Long: count = 0
    Dim i As Long
    For i = 2 To lastRow
        If Left(Trim(CStr(ws.Cells(i, 1).value)), Len(APIKEY_PREFIX)) = APIKEY_PREFIX Then
            count = count + 1
        End If
    Next i

    If count = 0 Then
        GetSavedApiKeys = emptyKeys
        Exit Function
    End If

    Dim keys() As String
    ReDim keys(count - 1)
    Dim k As Long: k = 0
    For i = 2 To lastRow
        If Left(Trim(CStr(ws.Cells(i, 1).value)), Len(APIKEY_PREFIX)) = APIKEY_PREFIX Then
            keys(k) = Trim(CStr(ws.Cells(i, 2).value))
            k = k + 1
        End If
    Next i

    GetSavedApiKeys = keys
End Function

Public Function IsHexKey32(s As String) As Boolean
    If Len(s) <> 32 Then Exit Function
    Dim i As Long
    For i = 1 To 32
        Dim c As String: c = mID(s, i, 1)
        If Not ((c >= "0" And c <= "9") Or (c >= "a" And c <= "f") Or (c >= "A" And c <= "F")) Then Exit Function
    Next i
    IsHexKey32 = True
End Function

' ============================================================
' DETECT PROVIDER FROM KEY FORMAT
' ============================================================
Public Function DetectProvider(apiKey As String) As String
    If IsHexKey32(apiKey) And GetAzureEndpoint() <> "" Then
        DetectProvider = "AZURE"
    ElseIf Left(apiKey, 4) = "csk-" Then
        DetectProvider = "CEREBRAS"
    ElseIf Left(apiKey, 11) = "github_pat_" Or Left(apiKey, 4) = "ghp_" Then
        DetectProvider = "GITHUB"
    ElseIf Left(apiKey, 4) = "gsk_" Then
        DetectProvider = "GROQ"
    ElseIf Left(apiKey, 6) = "nvapi-" Then
        DetectProvider = "NVIDIA"
    ElseIf Left(apiKey, 6) = "sk-or-" Then
        DetectProvider = "OPENROUTER"
    ElseIf Left(apiKey, 4) = "AIza" Then
        DetectProvider = "GEMINI"
    ElseIf Left(apiKey, 8) = "sk-proj-" Then
        DetectProvider = "OPENAI"
    ElseIf Left(apiKey, 3) = "sk-" Then
        DetectProvider = "OPENAI"
    Else
        DetectProvider = "GROQ"
    End If
End Function

Function ParseAIContent(jsonText As String) As String
    Dim doc As Object, choices As Object, message As Object, content As Object, parts As Object
    Dim item As Object, i As Long, result As String, reason As String
    Set doc = AIJsonParse(jsonText)
    Set choices = AIJsonChild(doc, "choices")
    Set item = AIJsonChild(choices, 1)
    reason = AIJsonText(AIJsonChild(item, "finish_reason"))
    Set message = AIJsonChild(item, "message")
    Set content = AIJsonChild(message, "content")
    result = AIJsonText(content)
    If result = "" Then
        For i = 1 To AIJsonCount(content)
            result = result & AIJsonText(AIJsonChild(AIJsonChild(content, i), "text"))
        Next i
    End If
    If result = "" Then result = AIJsonText(AIJsonChild(message, "refusal"))
    If result = "" Then
        Set item = AIJsonChild(AIJsonChild(doc, "candidates"), 1)
        Set parts = AIJsonChild(AIJsonChild(item, "content"), "parts")
        For i = 1 To AIJsonCount(parts)
            Set item = AIJsonChild(parts, i)
            If AIJsonText(AIJsonChild(item, "thought")) <> "true" Then
                result = result & AIJsonText(AIJsonChild(item, "text"))
            End If
        Next i
    End If
    If reason = "length" Then result = result & vbCrLf & "[Output limit reached. Ask for a shorter answer or continue.]"
    ParseAIContent = result
End Function
Function EscapeJson(s As String) As String
    Dim quoted As String
    quoted = AIJsonQuote(s)
    EscapeJson = Mid$(quoted, 2, Len(quoted) - 2)
End Function
Function ExtractJsonField(json As String, fieldName As String) As String
    Dim doc As Object
    Set doc = AIJsonParse(json)
    ExtractJsonField = AIJsonText(AIJsonChild(doc, fieldName))
End Function
Function AskAIWithImage(Prompt As String, model As String, _
                         Optional imgBase64 As String = "", _
                         Optional imgMime As String = "image/jpeg") As String
    Dim apiKey As String: apiKey = GetApiKey()
    If apiKey = "" Then AskAIWithImage = "": Exit Function

    Dim provider As String: provider = DetectProvider(apiKey)
    If model = "" Then
        Dim models() As String: models = GetProviderModels(provider)
        model = models(0)
    End If

    Select Case provider
        Case "GEMINI"
            AskAIWithImage = CallGeminiVision(Prompt, apiKey, model, imgBase64, imgMime)
        Case "GROQ", "NVIDIA", "OPENROUTER", "OPENAI", "AZURE", "CEREBRAS", "GITHUB"
            Dim URL As String
            Select Case provider
                Case "GROQ":       URL = "https://api.groq.com/openai/v1/chat/completions"
                Case "NVIDIA":     URL = "https://integrate.api.nvidia.com/v1/chat/completions"
                Case "OPENROUTER": URL = "https://openrouter.ai/api/v1/chat/completions"
                Case "OPENAI":     URL = "https://api.openai.com/v1/chat/completions"
                Case "CEREBRAS":   URL = "https://api.cerebras.ai/v1/chat/completions"
                Case "GITHUB":     URL = "https://models.inference.ai.azure.com/chat/completions"
                Case "AZURE":      URL = GetAzureEndpoint()
            End Select
            AskAIWithImage = CallOpenAICompatVision(Prompt, apiKey, URL, model, imgBase64, imgMime)
        Case Else
            AskAIWithImage = "{""error"":""Unknown provider""}"
    End Select
End Function

Private Function CallOpenAICompatVision(Prompt As String, apiKey As String, _
    URL As String, model As String, imgBase64 As String, imgMime As String) As String
    Dim contentVal As String, body As String
    contentVal = AIJsonQuote(Prompt)
    If imgBase64 <> "" Then
        contentVal = "[{""type"":""text"",""text"":" & AIJsonQuote(Prompt) & _
            "},{""type"":""image_url"",""image_url"":{""url"":" & _
            AIJsonQuote("data:" & imgMime & ";base64," & imgBase64) & "}}]"
    End If
    body = "{""model"":" & AIJsonQuote(model) & _
        ",""messages"":[{""role"":""user"",""content"":" & contentVal & "}]}"
    CallOpenAICompatVision = RequestJson("POST", URL, apiKey, body, 120)
End Function
Private Function CallGeminiVision(Prompt As String, apiKey As String, model As String, _
    imgBase64 As String, imgMime As String) As String
    Dim body As String, parts As String, URL As String
    URL = "https://generativelanguage.googleapis.com/v1beta/models/" & model & ":generateContent"
    parts = "{""text"":" & AIJsonQuote(Prompt) & "}"
    If imgBase64 <> "" Then
        parts = parts & ",{""inline_data"":{""mime_type"":" & AIJsonQuote(imgMime) & _
            ",""data"":" & AIJsonQuote(imgBase64) & "}}"
    End If
    body = "{""contents"":[{""parts"":[" & parts & "]}]}"
    CallGeminiVision = RequestJson("POST", URL, apiKey, body, 120)
End Function

                Case Else
                    ModelSupportsVision = False

                    ModelSupportsVision = True
                Case Else

                Case "meta/llama-4-scout-17b-16e-instruct", "meta/llama-4-maverick-17b-128e-instruct"
                    ModelSupportsVision = True

Private Function RequestJson(ByVal method As String, ByVal URL As String, _
    ByVal apiKey As String, ByVal body As String, ByVal timeoutSeconds As Long) As String
    On Error GoTo Failed
    If URL = "" Then Err.Raise 5, "AskAI", "Endpoint is not configured."
    If Left$(LCase$(URL), 8) <> "https://" Then Err.Raise 5, "AskAI", "HTTPS endpoint required."
    Set mHttp = CreateObject("MSXML2.ServerXMLHTTP.6.0")
    mHttp.setTimeouts 10000, 15000, 30000, timeoutSeconds * 1000
    mHttp.Open method, URL, True
    mHttp.setRequestHeader "Content-Type", "application/json; charset=utf-8"
    If apiKey <> "" Then
        If InStr(1, URL, "https://generativelanguage.googleapis.com/", vbTextCompare) = 1 Then
            mHttp.setRequestHeader "x-goog-api-key", apiKey
        ElseIf DetectProvider(apiKey) = "AZURE" Then
            mHttp.setRequestHeader "api-key", apiKey
        Else
            mHttp.setRequestHeader "Authorization", "Bearer " & apiKey
        End If
    End If
    Dim started As Single, elapsed As Single, raw As String, status As Long
    started = Timer
    If method = "GET" Then
        mHttp.send
    Else
        mHttp.send body
    End If
    Do
        If mCancelled Then Err.Raise vbObjectError + 2101, "AskAI", "Request cancelled."
        If mHttp.readyState = 4 Then Exit Do
        elapsed = Timer - started
        If elapsed < 0 Then elapsed = elapsed + 86400
        If elapsed > timeoutSeconds Then
            mHttp.abort
            Err.Raise vbObjectError + 2102, "AskAI", "Request timed out. Try again or choose another model."
        End If
        DoEvents
        AskAISleep 20
    Loop
    status = mHttp.status
    raw = mHttp.responseText
    Set mHttp = Nothing
    If status < 200 Or status >= 300 Then
        Dim detail As String
        detail = ApiErrorMessage(raw)
        If detail = "" Then detail = "The provider rejected the request."
        Select Case status
            Case 401, 403: detail = detail & " Check the key and model permissions."
            Case 404: detail = detail & " Click Refresh Models and select an available model."
            Case 429: detail = detail & " Rate limit or quota reached; wait before retrying."
        End Select
        Err.Raise vbObjectError + 2103, "AskAI", "HTTP " & status & ": " & detail
    End If
    RequestJson = raw
    Exit Function
Failed:
    Dim errorText As String, errorNumber As Long
    errorNumber = Err.Number
    errorText = Err.Description
    On Error Resume Next
    If Not mHttp Is Nothing Then mHttp.abort
    Set mHttp = Nothing
    On Error GoTo 0
    If apiKey <> "" Then errorText = Replace(errorText, apiKey, "[key hidden]")
    If mCancelled Then errorText = "Request cancelled."
    Err.Raise errorNumber, "AskAI", errorText
End Function

Private Function ApiErrorMessage(ByVal raw As String) As String
    On Error GoTo NotJson
    Dim doc As Object, apiError As Object
    Set doc = AIJsonParse(raw)
    Set apiError = AIJsonChild(doc, "error")
    If apiError Is Nothing Then Exit Function
    ApiErrorMessage = AIJsonText(AIJsonChild(apiError, "message"))
    If ApiErrorMessage = "" Then ApiErrorMessage = AIJsonText(apiError)
    If ApiErrorMessage = "" Then ApiErrorMessage = "Provider returned an error."
NotJson:
End Function

Private Sub RunAskAI()
    If mBusy Then Exit Sub
    Dim question As String, selectedModel As String, raw As String, answer As String, detail As String
    question = Trim$(txtQuestion.Text)
    If question = "" And m_ImageBase64 = "" Then
        MsgBox "Please enter a question or select an image.", vbInformation
        Exit Sub
    End If
    If GetApiKey() = "" Then
        MsgBox "Click the key field to enter an API key.", vbInformation
        Exit Sub
    End If
    If cboModel.ListIndex < 0 Then
        MsgBox "Select a model first.", vbInformation
        Exit Sub
    End If
    selectedModel = CStr(cboModel.Value)
    On Error GoTo Failed
    SaveSelectedModel selectedModel
    SetRequestBusy True
    lblRespone.Value = "AI processing..."
    raw = AskAIWithImage(question, selectedModel, m_ImageBase64, m_ImageMime)
    detail = ApiErrorMessage(raw)
    If detail <> "" Then Err.Raise 5, "AskAI", detail
    answer = ParseAIContent(raw)
    If answer = "" Then
        lblRespone.Value = "The provider returned no answer text. Try a shorter question or another model."
    Else
        lblRespone.Value = "AI: " & GetAIInfo(selectedModel) & vbCrLf & vbCrLf & answer
    End If
    GoTo Cleanup
Failed:
    detail = Err.Description
    If GetApiKey() <> "" Then detail = Replace(detail, GetApiKey(), "[key hidden]")
    lblRespone.Value = "AskAI: " & detail
Cleanup:
    SetRequestBusy False
    If mClosePending Then
        mClosePending = False
        Unload Me
    End If
End Sub

Private Function IsFreePrice(ByVal price As String) As Boolean
    Dim rx As Object
    Set rx = CreateObject("VBScript.RegExp")
    rx.Pattern = "^0(\.0+)?$"
    IsFreePrice = rx.Test(price)
End Function

Private Function GroqChatModel(ByVal modelId As String) As Boolean
    Dim lowerId As String
    lowerId = LCase$(modelId)
    If InStr(lowerId, "guard") > 0 Or InStr(lowerId, "whisper") > 0 Or InStr(lowerId, "orpheus") > 0 Then Exit Function
    GroqChatModel = (InStr(lowerId, "gpt-oss") > 0 Or InStr(lowerId, "llama") > 0 Or _
        InStr(lowerId, "qwen") > 0 Or InStr(lowerId, "minimax") > 0 Or InStr(lowerId, "compound") > 0)
End Function

Private Sub RefreshModels()
    Dim provider As String, URL As String, raw As String, savedModel As String
    Dim doc As Object, entries As Object, entry As Object, arch As Object, pricing As Object
    Dim newModels As Object, newVision As Object, modelId As String, include As Boolean, vision As Boolean
    Dim i As Long, modelKey As Variant, cache As String, cacheEntry As String, detail As String
    provider = DetectProvider(GetApiKey())
    If GetApiKey() = "" Then
        MsgBox "Enter an API key first.", vbInformation
        Exit Sub
    End If
    Select Case provider
        Case "GROQ": URL = "https://api.groq.com/openai/v1/models"
        Case "OPENROUTER": URL = "https://openrouter.ai/api/v1/models"
        Case Else
            MsgBox "Catalog refresh currently supports Groq and OpenRouter.", vbInformation
            Exit Sub
    End Select
    On Error GoTo Failed
    savedModel = CStr(cboModel.Value)
    SetRequestBusy True
    raw = RequestJson("GET", URL, GetApiKey(), "", 30)
    detail = ApiErrorMessage(raw)
    If detail <> "" Then Err.Raise 5, "AskAI", detail
    Set doc = AIJsonParse(raw)
    Set entries = AIJsonChild(doc, "data")
    Set newModels = CreateObject("Scripting.Dictionary")
    Set newVision = CreateObject("Scripting.Dictionary")
    For i = 1 To AIJsonCount(entries)
        Set entry = AIJsonChild(entries, i)
        modelId = AIJsonText(AIJsonChild(entry, "id"))
        include = (modelId <> "")
        vision = False
        If provider = "OPENROUTER" Then
            Set arch = AIJsonChild(entry, "architecture")
            include = include And AIJsonHasText(AIJsonChild(arch, "output_modalities"), "text")
            include = include And AIJsonHasText(AIJsonChild(arch, "input_modalities"), "text")
            vision = AIJsonHasText(AIJsonChild(arch, "input_modalities"), "image")
            If CBool(mFree.Value) Then
                Set pricing = AIJsonChild(entry, "pricing")
                include = include And IsFreePrice(AIJsonText(AIJsonChild(pricing, "prompt")))
                include = include And IsFreePrice(AIJsonText(AIJsonChild(pricing, "completion")))
                Dim requestPrice As String
                requestPrice = AIJsonText(AIJsonChild(pricing, "request"))
                If requestPrice <> "" Then include = include And IsFreePrice(requestPrice)
            End If
        Else
            include = include And GroqChatModel(modelId)
            If AIJsonText(AIJsonChild(entry, "active")) = "false" Then include = False
            vision = (modelId = "qwen/qwen3.8-27b")
        End If
        If include Then
            If Not newModels.Exists(modelId) Then newModels.Add modelId, modelId
            newVision(modelId) = vision
        End If
    Next i
    If newModels.Count = 0 Then Err.Raise 5, "AskAI", "No matching chat models returned. Previous list retained."
    mLoadingModels = True
    cboModel.Clear
    For Each modelKey In newModels.Keys
        cboModel.AddItem CStr(modelKey)
        cacheEntry = "0:" & CStr(modelKey)
        If newVision(modelKey) Then cacheEntry = "1:" & CStr(modelKey)
        If cache <> "" Then cache = cache & "|"
        cache = cache & cacheEntry
    Next modelKey
    Set mVision = newVision
    SelectModel savedModel
    mLoadingModels = False
    UpdateVisionButton
    If Len(cache) <= 30000 Then SaveAppSetting ModelCacheKey(provider), cache
    MsgBox CStr(newModels.Count) & " chat models loaded. Select a model before sending.", vbInformation, "Ask AI"
    GoTo Cleanup
Failed:
    mLoadingModels = False
    MsgBox "Refresh Models: " & Err.Description, vbExclamation
Cleanup:
    SetRequestBusy False
    If mClosePending Then
        mClosePending = False
        Unload Me
    End If
End Sub
